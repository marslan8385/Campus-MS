#include <iostream>
#include <fstream>
#include <string>
using namespace std;

class OverdueException
{
    int fine;

public:
    OverdueException(int f)
    {
        fine = f;
    }

    void showError()
    {
        cout << "OVERDUE ERROR! Fine = " << fine << " Rs\n";
    }
};

class LibraryItem
{
protected:
    int itemID;
    string title;
    string author;
    int publicationYear;

public:
    virtual void getdata()
    {
        cout << "Item ID: ";
        cin >> itemID;

        cout << "Title: ";
        cin >> title;

        cout << "Author: ";
        cin >> author;

        cout << "Publication Year: ";
        cin >> publicationYear;
    }

    virtual void display() = 0;

    string getTitle()
    {
        return title;
    }

    int getID()
    {
        return itemID;
    }

    virtual void save(ofstream &out)
    {
        out << itemID << " " << title << " " << author << " " << publicationYear << " ";
    }

    virtual void load(ifstream &in)
    {
        in >> itemID >> title >> author >> publicationYear;
    }
};

class Book : public LibraryItem
{
private:
    string ISBN;
    string genre;
    int copiesAvailable;

public:
    void getdata() override
    {
        LibraryItem::getdata();

        cout << "ISBN: ";
        cin >> ISBN;

        cout << "Genre: ";
        cin >> genre;

        cout << "Copies Available: ";
        cin >> copiesAvailable;
    }

    void display() override
    {
        cout << "\n--- BOOK ---\n";
        cout << "ID: " << itemID << endl;
        cout << "Title: " << title << endl;
        cout << "Author: " << author << endl;
        cout << "Year: " << publicationYear << endl;
        cout << "ISBN: " << ISBN << endl;
        cout << "Genre: " << genre << endl;
        cout << "Copies: " << copiesAvailable << endl;
    }

    void save(ofstream &out) override
    {
        out << "BOOK ";
        LibraryItem::save(out);
        out << ISBN << " " << genre << " " << copiesAvailable << endl;
    }

    void load(ifstream &in) override
    {
        LibraryItem::load(in);
        in >> ISBN >> genre >> copiesAvailable;
    }
};

class Journal : public LibraryItem
{
private:
    string ISSN;
    int volume;
    int issueNumber;

public:
    void getdata() override
    {
        LibraryItem::getdata();

        cout << "ISSN: ";
        cin >> ISSN;

        cout << "Volume: ";
        cin >> volume;

        cout << "Issue Number: ";
        cin >> issueNumber;
    }

    void display() override
    {
        cout << "\n--- JOURNAL ---\n";
        cout << "ID: " << itemID << endl;
        cout << "Title: " << title << endl;
        cout << "Author: " << author << endl;
        cout << "Year: " << publicationYear << endl;
        cout << "ISSN: " << ISSN << endl;
        cout << "Volume: " << volume << endl;
        cout << "Issue: " << issueNumber << endl;
    }

    void save(ofstream &out) override
    {
        out << "JOURNAL ";
        LibraryItem::save(out);
        out << ISSN << " " << volume << " " << issueNumber << endl;
    }

    void load(ifstream &in) override
    {
        LibraryItem::load(in);
        in >> ISSN >> volume >> issueNumber;
    }
};

struct IssueRecord
{
    string rollNo;
    int itemID;
};

class Library
{
private:
    LibraryItem* catalog[100];
    int count;

    IssueRecord issued[100];
    int issueCount;

public:
    Library()
    {
        count = 0;
        issueCount = 0;
    }

    void addItem(LibraryItem* item)
    {
        catalog[count++] = item;
    }

    void searchByTitle(string t)
    {
        bool found = false;

        for (int i = 0; i < count; i++)
        {
            if (catalog[i]->getTitle() == t)
            {
                catalog[i]->display();
                found = true;
            }
        }

        if (!found)
            cout << "Item not found!\n";
    }
    void issueItem(string roll, int id)
    {
        issued[issueCount].rollNo = roll;
        issued[issueCount].itemID = id;
        issueCount++;
    }
    void checkOverdue(int days)
    {
        try
        {
            if (days > 14)
            {
                throw OverdueException((days - 14) * 50);
            }
            else
            {
                cout << "No overdue items.\n";
            }
        }
        catch (OverdueException e)
        {
            e.showError();
        }
    }

    void saveToFile()
    {
        ofstream out("library.txt");

        out << count << endl;

        for (int i = 0; i < count; i++)
        {
            catalog[i]->save(out);
        }

        out.close();
        cout << "Library saved successfully!\n";
    }

    void loadFromFile()
    {
        ifstream in("library.txt");

        if (!in)
        {
            cout << "File not found!\n";
            return;
        }

        in >> count;

        for (int i = 0; i < count; i++)
        {
            string type;
            in >> type;

            if (type == "BOOK")
            {
                catalog[i] = new Book();
            }
            else
            {
                catalog[i] = new Journal();
            }

            catalog[i]->load(in);
        }

        in.close();
        cout << "Library loaded successfully!\n";
    }

    void displayAll()
    {
        for (int i = 0; i < count; i++)
        {
            catalog[i]->display();
        }
    }
};

int main()
{
    Library lib;

    int choice;

    do
    {
        cout << "\n===== LIBRARY SYSTEM =====\n";
        cout << "1. Add Book\n";
        cout << "2. Add Journal\n";
        cout << "3. Search by Title\n";
        cout << "4. Display All\n";
        cout << "5. Save to File\n";
        cout << "6. Load from File\n";
        cout << "7. Check Overdue\n";
        cout << "0. Exit\n";
        cout << "Choice: ";
        cin >> choice;

        system("cls");

        switch (choice)
        {
        case 1:
        {
            Book* b = new Book();
            b->getdata();
            lib.addItem(b);
            break;
        }

        case 2:
        {
            Journal* j = new Journal();
            j->getdata();
            lib.addItem(j);
            break;
        }

        case 3:
        {
            string t;
            cout << "Enter title: ";
            cin >> t;
            lib.searchByTitle(t);
            break;
        }

        case 4:
            lib.displayAll();
            break;

        case 5:
            lib.saveToFile();
            break;

        case 6:
            lib.loadFromFile();
            break;

        case 7:
        {
            int d;
            cout << "Enter days: ";
            cin >> d;
            lib.checkOverdue(d);
            break;
        }
        }

    } while (choice != 0);

    return 0;
}
